cmake_minimum_required(VERSION 3.24)
project(AnrTests LANGUAGES C CXX)
set(NATIVE "${CMAKE_CURRENT_LIST_DIR}/../../source/native")
set(STREAMLINE_ROOT "" CACHE PATH "Pinned Streamline SDK 2.14.1")
set(NGX_INCLUDE "${STREAMLINE_ROOT}/external/ngx-sdk/include")

add_executable(AnrRulesTests anr_rules_tests.cpp)
set(ANR_TARGETS AnrRulesTests)

if(EXISTS "${NGX_INCLUDE}/nvsdk_ngx_defs.h")
    foreach(fake IN ITEMS FakeAnrProvider:anr_provider FakeNvidiaProvider:nvidia_provider
            FakeNoEvaluateProvider:no_evaluate_provider)
        string(REPLACE ":" ";" parts "${fake}")
        list(GET parts 0 name)
        list(GET parts 1 definition)
        add_library(${name} SHARED fake_provider.cpp "${CMAKE_CURRENT_LIST_DIR}/${definition}.def")
        target_include_directories(${name} PRIVATE "${NGX_INCLUDE}")
        list(APPEND ANR_TARGETS ${name})
    endforeach()
    add_executable(AnrModuleTests anr_module_tests.cpp "${NATIVE}/anr/anr.cpp")
    target_include_directories(AnrModuleTests PRIVATE "${NGX_INCLUDE}" "${STREAMLINE_ROOT}/include")
    add_dependencies(AnrModuleTests FakeAnrProvider FakeNvidiaProvider FakeNoEvaluateProvider)
    list(APPEND ANR_TARGETS AnrModuleTests)
else()
    message(STATUS "AnrModuleTests skipped: set STREAMLINE_ROOT")
endif()

foreach(target ${ANR_TARGETS})
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${NATIVE}" "${CMAKE_CURRENT_LIST_DIR}")
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    set_target_properties(${target} PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()

enable_testing()
add_test(NAME AnrRulesTests COMMAND AnrRulesTests)
if(TARGET AnrModuleTests)
    foreach(mode not-requested refused-xmx translated-gpu translated-gpu-later no-provider nvidia-provider no-evaluate
            create nvidia-loaded-first stale-copy other-feature second-device detached-list init-fails create-fails evaluate release limit)
        string(REPLACE "-" "_" name "AnrModule_${mode}")
        add_test(NAME ${name} COMMAND AnrModuleTests ${mode} "${CMAKE_CURRENT_BINARY_DIR}")
    endforeach()
endif()
