cmake_minimum_required(VERSION 3.24)
project(XmxTests LANGUAGES C CXX)
set(NATIVE "${CMAKE_CURRENT_LIST_DIR}/../../source/native")
set(STREAMLINE_ROOT "" CACHE PATH "Pinned Streamline SDK 2.14.1")
set(XESS_SDK_ROOT "" CACHE PATH "Pinned XeSS SDK 3.0.2")
set(MFG_GENERATED_DIR "" CACHE PATH "The DLL build's generated folder (tools/linux/build.sh)")
set(MFG_PAYLOAD_DLL "" CACHE FILEPATH "The built MFGUnlocker.dll, the embedded programs' module")
set(XMX_RUNTIME_DIR "" CACHE PATH "Folder with libxess_fg.dll 1.3.1.78 and libxell.dll")
add_executable(XmxPolicyTests xmx_policy_tests.cpp)
add_executable(XmxRulesTests xmx_rules_tests.cpp "${NATIVE}/xmx/xmx_context.cpp")
add_executable(XmxControlCarryTests control_carry_tests.cpp)
add_executable(XmxWindowStatsTests window_stats_tests.cpp)
add_executable(XmxTagStatesTests xmx_tag_states_tests.cpp)
add_executable(XmxReflexSleepTests reflex_sleep_tests.cpp)
add_executable(XmxStandInTests xmx_stand_in_tests.cpp)
add_executable(XmxLowLatencyRulesTests low_latency_rules_tests.cpp)
add_executable(XmxUiStateTests xmx_ui_state_tests.cpp)
add_executable(XmxLowLatencyHookTests low_latency_hook_tests.cpp "${NATIVE}/xmx/low_latency_hold.cpp")
set(XMX_TARGETS XmxPolicyTests XmxRulesTests XmxControlCarryTests XmxWindowStatsTests XmxTagStatesTests XmxReflexSleepTests XmxStandInTests XmxLowLatencyRulesTests XmxLowLatencyHookTests XmxUiStateTests)
if(EXISTS "${XESS_SDK_ROOT}/inc/xell/xell.h")
    add_executable(XmxLatencyTests xmx_latency_tests.cpp "${NATIVE}/xmx/xmx_latency.cpp")
    target_include_directories(XmxLatencyTests PRIVATE "${XESS_SDK_ROOT}/inc/xell")
    target_link_libraries(XmxLatencyTests PRIVATE d3d12 dxgi)
    list(APPEND XMX_TARGETS XmxLatencyTests)
endif()
if(EXISTS "${XESS_SDK_ROOT}/inc/xess_fg/xefg_swapchain_d3d12.h" AND EXISTS "${STREAMLINE_ROOT}/include/sl.h"
        AND EXISTS "${MFG_GENERATED_DIR}/embedded_xess_programs.generated.h")
    add_executable(XmxRuntimeTests xmx_runtime_tests.cpp "${NATIVE}/xmx/xmx_context.cpp" "${NATIVE}/xess/xess_mod.cpp"
        "${NATIVE}/xess/host_unlock.cpp" "${NATIVE}/xess/runtime_payload.cpp" "${NATIVE}/xess/xess_runtime.cpp"
        "${NATIVE}/embedded_payload.cpp" "${NATIVE}/third_party/lzma/LzmaDec.c")
    target_include_directories(XmxRuntimeTests PRIVATE "${MFG_GENERATED_DIR}" "${STREAMLINE_ROOT}/include"
        "${STREAMLINE_ROOT}/external/ngx-sdk/include" "${XESS_SDK_ROOT}/inc/xess_fg" "${XESS_SDK_ROOT}/inc/xell")
    target_compile_definitions(XmxRuntimeTests PRIVATE MFG_UNLOCK_XESS=1)
    target_link_libraries(XmxRuntimeTests PRIVATE d3d12 dxgi bcrypt)
    target_link_options(XmxRuntimeTests PRIVATE /FORCE:UNRESOLVED)
    list(APPEND XMX_TARGETS XmxRuntimeTests)
    add_executable(XmxModuleTests xmx_module_tests.cpp "${NATIVE}/xmx/xmx.cpp" "${NATIVE}/xmx/xmx_latency.cpp"
        "${NATIVE}/xmx/low_latency_hold.cpp")
    target_include_directories(XmxModuleTests PRIVATE "${STREAMLINE_ROOT}/include"
        "${STREAMLINE_ROOT}/external/ngx-sdk/include" "${XESS_SDK_ROOT}/inc/xell")
    target_compile_definitions(XmxModuleTests PRIVATE MFG_UNLOCK_XESS=1)
    target_link_libraries(XmxModuleTests PRIVATE d3d12 dxgi)
    list(APPEND XMX_TARGETS XmxModuleTests)
    add_executable(XmxGeneratorTests xmx_generator_tests.cpp "${NATIVE}/xmx/xmx_generator.cpp")
    target_include_directories(XmxGeneratorTests PRIVATE "${STREAMLINE_ROOT}/external/ngx-sdk/include")
    target_link_libraries(XmxGeneratorTests PRIVATE d3d12 dxgi)
    list(APPEND XMX_TARGETS XmxGeneratorTests)
else()
    message(STATUS "XmxRuntimeTests skipped: set STREAMLINE_ROOT, XESS_SDK_ROOT and MFG_GENERATED_DIR")
endif()
foreach(target ${XMX_TARGETS})
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${NATIVE}")
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    set_target_properties(${target} PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
enable_testing()
add_test(NAME XmxPolicyTests COMMAND XmxPolicyTests)
add_test(NAME XmxRulesTests COMMAND XmxRulesTests)
add_test(NAME XmxControlCarryTests COMMAND XmxControlCarryTests)
add_test(NAME XmxWindowStatsTests COMMAND XmxWindowStatsTests)
add_test(NAME XmxTagStatesTests COMMAND XmxTagStatesTests)
add_test(NAME XmxReflexSleepTests COMMAND XmxReflexSleepTests)
add_test(NAME XmxStandInTests COMMAND XmxStandInTests)
add_test(NAME XmxLowLatencyRulesTests COMMAND XmxLowLatencyRulesTests)
add_test(NAME XmxUiStateTests COMMAND XmxUiStateTests)
foreach(mode not-available forwarding install-with-hold-on second-device)
    string(REPLACE "-" "_" name "XmxLowLatencyHook_${mode}")
    add_test(NAME ${name} COMMAND XmxLowLatencyHookTests ${mode})
endforeach()
if(TARGET XmxLatencyTests)
    add_test(NAME XmxLatencyTests COMMAND XmxLatencyTests fakes)
    add_test(NAME XmxLatencyRuntimeTests COMMAND XmxLatencyTests runtime "${XMX_RUNTIME_DIR}")
    set_tests_properties(XmxLatencyRuntimeTests PROPERTIES SKIP_RETURN_CODE 2)
endif()
if(TARGET XmxRuntimeTests)
    add_test(NAME XmxRuntimeModelTests COMMAND XmxRuntimeTests "${XMX_RUNTIME_DIR}" "${MFG_PAYLOAD_DLL}" model)
    add_test(NAME XmxRuntimeDeviceTests COMMAND XmxRuntimeTests "${XMX_RUNTIME_DIR}" "${MFG_PAYLOAD_DLL}" device)
    set_tests_properties(XmxRuntimeDeviceTests PROPERTIES SKIP_RETURN_CODE 2)
    foreach(mode route-before-latch no-route routes-without-latch hooked-without-latch evaluate-entry translated-gpu translated-gpu-first
            standalone-keeps-its-reason native-gpu loaded-runtime folder-runtime loaded-not-fg78
            stand-in-key-off stand-in-not-latched stand-in-no-runtime stand-in-create stand-in-route-failed stand-in-evaluate
            stand-in-release stand-in-limit)
        string(REPLACE "-" "_" name "XmxModule_${mode}")
        add_test(NAME ${name} COMMAND XmxModuleTests ${mode} "${XMX_RUNTIME_DIR}")
    endforeach()
endif()
if(TARGET XmxGeneratorTests)
    add_test(NAME XmxGeneratorMemoryEmptyTests COMMAND XmxGeneratorTests memory-empty)
    add_test(NAME XmxGeneratorReuseTests COMMAND XmxGeneratorTests reuse)
    add_test(NAME XmxGeneratorFramesTests COMMAND XmxGeneratorTests frames)
    add_test(NAME XmxGeneratorRefusedTests COMMAND XmxGeneratorTests refused)
    add_test(NAME XmxGeneratorRefusedAlternatingTests COMMAND XmxGeneratorTests refused-alternating)
    add_test(NAME XmxGeneratorFallbackTests COMMAND XmxGeneratorTests fallback)
    add_test(NAME XmxGeneratorMemoryDeviceTests COMMAND XmxGeneratorTests memory-device)
    add_test(NAME XmxGeneratorCopyMismatchTests COMMAND XmxGeneratorTests copy-mismatch)
    set_tests_properties(XmxGeneratorCopyMismatchTests XmxGeneratorReuseTests XmxGeneratorFramesTests XmxGeneratorRefusedTests XmxGeneratorRefusedAlternatingTests XmxGeneratorFallbackTests XmxGeneratorMemoryDeviceTests PROPERTIES SKIP_RETURN_CODE 2)
endif()
